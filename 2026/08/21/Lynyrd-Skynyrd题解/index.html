Lynyrd Skynyrd题解

题目地址: https://codeforces.com/contest/1143/E

题解用的是倍增
写了一个转换成图的写法
时间复杂度比题解快

#include <bits/stdc++.h>
#define all(x) x.begin(), x.end()
#define int long long
using namespace std;

const int mod = 998244353;
#define double long double
using i128 = __int128_t;
typedef pair<int, int> pii;

// reverseGraph[x]：所有满足 next[i] = x 的位置 i
vector<vector<int>> reverseGraph(200010);

// earliestEnd[i]：从位置 i 开始，形成长度 n 的合法子序列时的结束位置
vector<int> earliestEnd(200010);

// path[pos]：DFS 当前路径上第 pos 个位置
vector<int> path(200010);

// visited[i]：位置 i 是否已经被 DFS 处理
vector<int> visited(200010);

// pos[x]：数字 x 在排列 p 中的下标（0-based）
vector<int> pos(200010);
int n;
/*
    x:当前点的位置
    left, right：当前路径中有效滑动窗口的左右端点
    如果当前路径长度 >= n，
    那么就找到了一个长度为 n 的循环移位子序列。
*/
void dfs(int x, int left, int right) {
    //只保留长度为 n 的滑动窗口
    //path[left ... right]
    if (right - left + 1 >= n) {
        left++;
    }
    right++;
    for (auto nextNode : reverseGraph[x]) {

        if (visited[nextNode])
            continue;
        visited[nextNode] = 1;

        // 当前路径已经有至少 n 个节点
        if (right - left + 1 >= n) {
            earliestEnd[nextNode] = path[left];
        }
        path[right] = nextNode;
        dfs(nextNode, left, right);
    }
}
void solve() {
    int m, q;
    cin >> n >> m >> q;
    if (n > m) {
        while (q--)
        cout << 0;
        return;
    }

    // p：长度为 n 的排列
    // a：长度为 m 的数组
    vector<int> p(n + 1);
    vector<int> a(m + 1);

    // nextPosition[i]：
    // 从 a[i] 出发，按照 p 的循环顺序，
    // 下一个可以接上的位置
    vector<int> nextPosition(m + 1);

    // lastPosition[rank]：
    // 当前从右往左扫描 a 时，
    // p 中第 rank 个位置对应的数字最近出现在哪里
    vector<int> lastPosition(n + 1);

    for (int i = 1; i <= n; i++) {
        cin >> p[i];
        //p中数字p[i]的0-based位置
        pos[p[i]] = i - 1;
    }
    for (int i = 1; i <= m; i++)cin >> a[i];
    lastPosition[pos[a[m]]] = m;

    for (int i = m - 1; i >= 1; i--) {
        int rank = pos[a[i]];
        int nextRank = (rank + 1) % n;
        if (lastPosition[nextRank]) {
            nextPosition[i] = lastPosition[nextRank];
            //从右往左建立 nextPosition。
            //即寻找这个元素在 a[i] 后面最近出现的位置。
        }
        lastPosition[rank] = i;
    }
    for (int i = 1; i <= m; i++) {
        reverseGraph[nextPosition[i]].push_back(i);
        //从后往前 DFS，建立反向图
    }

    //对每一条链进行 DFS。
    for (int i = m; i >= 1; i--) {
        if (visited[i])continue;
        visited[i] = 1;
        path[0] = i;
        dfs(i, 0, 0);
        //DFS 可以一次处理整条链，不需要像题解那样对每个节点单独 binary lifting。
    }
    int minEnd = 0;
    //earliestEnd[i] 原本表示：从 i 开始的合法子序列最早在哪里结束。
    //所以需要做后缀最小值
    for (int i = m; i >= 1; i--) {
        if (minEnd == 0) {
            minEnd = earliestEnd[i];
            continue;
        }
        if (earliestEnd[i])
            minEnd = min(minEnd, earliestEnd[i]);
        earliestEnd[i] = minEnd;
    }
    if (n == 1) {
        for (int i = 1; i <= m; i++)
            earliestEnd[i] = i;
    }
    while (q--) {
        int left, right;
        cin >> left >> right;
        if (earliestEnd[left] == 0) {
            cout << 0;
            continue;
        }
        if (right >= earliestEnd[left])
            cout << 1;
        else
            cout << 0;
    }
}
signed main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
```cpp
时间复杂度O(n+m+p)
题解时间复杂度O((n+m)logn +p)
